# The repository-root CMake project selects this desktop build.
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(FV_SHARED "${CMAKE_CURRENT_SOURCE_DIR}/../shared")
set(FV_PICO "${CMAKE_CURRENT_SOURCE_DIR}/../pico")
option(FV_SANITIZERS "Enable desktop address/undefined sanitizers" OFF)
option(FV_ENABLE_FIDO_ENGINE "Build portable pico-fido engine (AGPLv3)" OFF)
if(FV_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

foreach(module crypto storage security benchmark ui)
  add_subdirectory(${FV_SHARED}/${module} ${CMAKE_BINARY_DIR}/shared/${module})
endforeach()

if(FV_BUILD_TESTS)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  include(tests.cmake)
endif()

if(FV_ENABLE_FIDO_ENGINE)
  add_subdirectory(${FV_SHARED}/fido ${CMAKE_BINARY_DIR}/shared/fido)
  if(FV_BUILD_TESTS)
    include(fido_tests.cmake)
  endif()
endif()
