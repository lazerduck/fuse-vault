cmake_minimum_required(VERSION 3.20)

  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

if(FV_BUILD_TESTS)
  function(fv_mock_pico_hmac target)
    target_sources(${target} PRIVATE src/crypto/hmac.c src/crypto/hmac_self_test.c
      src/crypto/backends/pico_hmac.c tests/sha_sdk_fake.c)
    target_include_directories(${target} PRIVATE tests/sha_stubs)
    target_compile_definitions(${target} PRIVATE FV_HMAC_PICO=1)
  endfunction()
  add_executable(hmac_pico_tests tests/hmac_pico_tests.c)
  fv_mock_pico_hmac(hmac_pico_tests)
  target_link_libraries(hmac_pico_tests PRIVATE fv_crypto)
  add_test(NAME pico_hmac_contract COMMAND hmac_pico_tests)
  add_executable(auth_store_pico_tests tests/auth_store_tests.c)
  fv_mock_pico_hmac(auth_store_pico_tests)
  target_link_libraries(auth_store_pico_tests PRIVATE fv_storage OpenSSL::Crypto)
  add_test(NAME pico_hmac_storage COMMAND auth_store_pico_tests)
  add_executable(benchmark_peer_pico tests/benchmark_peer.c)
  fv_mock_pico_hmac(benchmark_peer_pico)
  target_link_libraries(benchmark_peer_pico PRIVATE fv_benchmark)
  add_test(NAME pico_hmac_pipeline COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/board_bench_tests.py $<TARGET_FILE:benchmark_peer_pico>)
endif()
